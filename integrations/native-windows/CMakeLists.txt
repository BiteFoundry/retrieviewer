cmake_minimum_required(VERSION 3.21)
project(retrieviewer_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
get_filename_component(PET_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${PET_ROOT}/dist/native-memory/quickjs-src" quickjs-build EXCLUDE_FROM_ALL)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${PET_ROOT}/dist/native-memory/zlib-src" zlib-build EXCLUDE_FROM_ALL)
add_executable(retrieviewer-native host.cpp)
target_link_options(retrieviewer-native PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/application.manifest")
target_link_libraries(retrieviewer-native PRIVATE qjs qjs-libc zlibstatic gdiplus user32 gdi32 advapi32 psapi)
target_compile_definitions(retrieviewer-native PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX UNICODE _UNICODE)
