cmake_minimum_required(VERSION 3.21)
project(retrieviewer_memory_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
get_filename_component(PET_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${PET_ROOT}/dist/native-memory/quickjs-src" quickjs-build EXCLUDE_FROM_ALL)
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${PET_ROOT}/dist/native-memory/zlib-src" zlib-build EXCLUDE_FROM_ALL)
add_executable(pet-probe host.cpp)
target_link_libraries(pet-probe PRIVATE qjs qjs-libc zlibstatic gdiplus user32 gdi32)
target_compile_definitions(pet-probe PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX UNICODE _UNICODE)
